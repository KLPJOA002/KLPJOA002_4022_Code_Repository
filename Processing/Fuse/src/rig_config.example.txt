# Rig geometry for point cloud fusion - edit and re-run, no rebuild needed.
# Load with: ./fuse_scans --rig-config rig_config.example.txt --sessions ...
#
# Any of these can still be overridden per-run with the matching CLI flag
# (--tilt-axis, --tilt-sign, --rotation-axis, --rotation-step-deg,
# --rotation-sign) without editing this file.

# --- Tilt (within a session, from each subfolder's "_a<deg>" name) ---
tilt_axis = x
tilt_sign = 1

# Where the tilt axis passes through, in the camera's frame, in metres.
# The camera sits off the tilt head's pivot, so tilting swings it through an
# arc; ignoring that leaves a seed error growing with sin(tilt), which is
# what makes high-tilt scans fail to register while low ones look perfect.
# Leave this commented out and the pipeline measures it from your low-tilt
# scans on every run (and prints what it found). Set it only to pin a value
# you have measured yourself - doing so skips the calibration stage.
# tilt_pivot = 0, -0.04, -0.12

# Same for the turntable axis. Not auto-estimated; leave at zero unless you
# know it.
# rotation_pivot = 0, 0, 0

# How the IMU's axes sit relative to the frame the PLYs are written in:
# identity, rx180, ry180, rz180, or auto to let the tilt calibration choose.
# A fixed property of the camera (rx180 for the D456 exports used here:
# y and z negated). "auto" can pick wrongly - see README, "The IMU is not
# necessarily in the point cloud's frame". Overridden by --imu-frame.
imu_frame = rx180

# --- Rotation (between sessions) ---
# Axis your rig's rotation stage turns about.
rotation_axis = z
# Assumed degrees between consecutive sessions, in the order you pass them
# on the command line - only used as a fallback for sessions whose folder
# name doesn't itself contain a "_r<deg>" token.
rotation_step_deg = 45
rotation_sign = 1

# How tilt and rotation compose. true: world = R_rotation * R_tilt (tilt
# happens in the frame that then gets rotated - tilt head mounted on a
# turntable). Flip to false if alignment quality is consistently poor with
# these settings on, before assuming the angles themselves are wrong.
compose_tilt_first = true
